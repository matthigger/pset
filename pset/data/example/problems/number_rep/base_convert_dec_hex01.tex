\begin{enumerate}[i]

    \item Convert the number 635 from decimal to binary using Euclid's Division Algorithm.

    \sol{
        \begin{align*}
            635 = 317 * 2 + 1& \\
            317 = 158* 2 +1& \\
            158 = 79* 2 +0& \\
            79 = 39* 2 +1& \\
            39 = 19* 2 +1& \\
            19 = 9* 2 +1& \\
            9 = 4* 2 +1& \\
            4 = 2* 2 +0& \\
            2 = 1* 2 +0& \\
            1 = 0* 2 +1&
        \end{align*}
        so that $635 = (1001111011)_2$
    }
    \rub{

            \item 3 pt correct final answer (-1 per bit to min 0)
            \item 3 pt work shown via Euclid's Division Algorithm which clearly yields final answer
        \end{itemize}
